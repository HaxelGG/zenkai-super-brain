**Architecture:** Pure date/aggregation helpers (`summary.ts`, TDD) + an isolated Shopify connector (`client.ts` reads the owner-only token from the vault and POSTs to the Admin GraphQL API; `queries.ts` shapes the data and distinguishes not-configured / no-access / data / error) + a server-rendered `/tienda` page. **Tech Stack:** Next.js 16 (server components), Supabase (RLS, owner-only secrets), Shopify Admin GraphQL API, TypeScript, Vitest, Tailwind v4 / Atelier tokens.

```ts
export type OrderRow = { name: string; createdAt: string; financialStatus: string; total: number; currency: string; customer: string | null };
export type ProductRow = { title: string; status: string; inventory: number; price: number };

/** Sum order totals whose createdAt date (YYYY-MM-DD prefix) is within [fromISO, toISO]. */
export function salesInRange(orders: OrderRow[], fromISO: string, toISO?: string): number {
  return orders.reduce((sum, o) => {
    const d = o.createdAt.slice(0, 10);
    if (d < fromISO) return sum;
    if (toISO && d > toISO) return sum;
    return sum + o.total;
  }, 0);
}

export function ordersCount(orders: OrderRow[], fromISO: string): number {
  return orders.filter((o) => o.createdAt.slice(0, 10) >= fromISO).length;
}

export function lowStock(products: ProductRow[], threshold = 0): ProductRow[] {
  return products.filter((p) => p.status === "ACTIVE" && p.inventory <= threshold);
}

export function todayISO(today: Date): string {
  return today.toISOString().slice(0, 10);
}

export function firstOfMonthISO(today: Date): string {
  return `${today.toISOString().slice(0, 7)}-01`;
}
```

```ts
import { createClient } from "@/lib/supabase/server";

export const SHOPIFY_API_VERSION = "2025-01";

export type ShopifyCreds = { domain: string; token: string };

/** Reads the Shopify domain + Admin token from the Conexiones vault for this company.
 * integration_secrets is owner-only (RLS), so a non-owner session yields null. */
export async function getShopifyCreds(companyId: string): Promise<ShopifyCreds | null> {
  const supabase = await createClient();
  const { data: conn } = await supabase
    .from("integration_connections")
    .select("id")
    .eq("company_id", companyId)
    .eq("provider", "shopify")
    .maybeSingle();
  if (!conn) return null;
  const { data: secrets } = await supabase
    .from("integration_secrets")
    .select("name,value")
    .eq("connection_id", conn.id);
  if (!secrets || secrets.length === 0) return null;
  const map = new Map(secrets.map((s) => [s.name, s.value]));
  const domain = map.get("SHOPIFY_SHOP_DOMAIN");
  const token = map.get("SHOPIFY_ADMIN_TOKEN");
  if (!domain || !token) return null;
  return { domain: domain.replace(/^https?:\/\//, "").replace(/\/$/, ""), token };
}

export async function shopifyGraphQL<T = unknown>(
  creds: ShopifyCreds,
  query: string,
  variables?: Record<string, unknown>,
): Promise<{ data: T } | { error: string }> {
  try {
    const res = await fetch(`https://${creds.domain}/admin/api/${SHOPIFY_API_VERSION}/graphql.json`, {
      method: "POST",
      headers: { "X-Shopify-Access-Token": creds.token, "Content-Type": "application/json" },
      body: JSON.stringify({ query, variables: variables ?? {} }),
      cache: "no-store",
    });
    if (!res.ok) return { error: `Shopify ${res.status}: ${(await res.text()).slice(0, 200)}` };
    const json = await res.json();
    if (json.errors) return { error: (json.errors as { message: string }[]).map((e) => e.message).join("; ") };
    return { data: json.data as T };
  } catch (e) {
    return { error: e instanceof Error ? e.message : "Error de red con Shopify." };
  }
}
```

Create `src/lib/shopify/queries.ts`:

```ts
import { createClient } from "@/lib/supabase/server";
import { getShopifyCreds, shopifyGraphQL } from "./client";
import { firstOfMonthISO, type OrderRow, type ProductRow } from "./summary";

export type StoreData =
  | { configured: false }
  | { configured: true; access: false }
  | { configured: true; access: true; shopName: string; currency: string; orders: OrderRow[]; products: ProductRow[] }
  | { error: string };

const STORE_QUERY = `query Store($q: String!) {
  shop { name currencyCode }
  orders(first: 50, sortKey: CREATED_AT, reverse: true, query: $q) {
    edges { node {
      name
      createdAt
      displayFinancialStatus
      currentTotalPriceSet { shopMoney { amount } }
      customer { displayName }
    } }
  }
  products(first: 25, sortKey: TITLE) {
    edges { node { title status totalInventory variants(first: 1) { edges { node { price } } } } }
  }
}`;

type GqlResp = {
  shop: { name: string; currencyCode: string };
  orders: { edges: { node: {
    name: string; createdAt: string; displayFinancialStatus: string | null;
    currentTotalPriceSet: { shopMoney: { amount: string } } | null;
    customer: { displayName: string } | null;
  } }[] };
  products: { edges: { node: {
    title: string; status: string; totalInventory: number | null;
    variants: { edges: { node: { price: string } }[] };
  } }[] };
};

export async function getStoreData(companyId: string): Promise<StoreData> {
  const supabase = await createClient();
  const { data: conn } = await supabase
    .from("integration_connections")
    .select("status")
    .eq("company_id", companyId)
    .eq("provider", "shopify")
    .maybeSingle();
  if (!conn || conn.status !== "connected") return { configured: false };

  const creds = await getShopifyCreds(companyId);
  if (!creds) return { configured: true, access: false };

  const monthStart = firstOfMonthISO(new Date());
  const res = await shopifyGraphQL<GqlResp>(creds, STORE_QUERY, { q: `created_at:>=${monthStart}` });
  if ("error" in res) return { error: res.error };

  const d = res.data;
  const orders: OrderRow[] = d.orders.edges.map((e) => ({
    name: e.node.name,
    createdAt: e.node.createdAt,
    financialStatus: e.node.displayFinancialStatus ?? "—",
    total: Number(e.node.currentTotalPriceSet?.shopMoney?.amount ?? 0),
    currency: d.shop.currencyCode,
    customer: e.node.customer?.displayName ?? null,
  }));
  const products: ProductRow[] = d.products.edges.map((e) => ({
    title: e.node.title,
    status: e.node.status,
    inventory: e.node.totalInventory ?? 0,
    price: Number(e.node.variants.edges[0]?.node.price ?? 0),
  }));

  return { configured: true, access: true, shopName: d.shop.name, currency: d.shop.currencyCode, orders, products };
}
```

```tsx
import type { ReactNode } from "react";
import { getActiveCompany } from "@/lib/active-company";
import { createClient } from "@/lib/supabase/server";
import { PageHeader } from "@/components/atelier/page-header";
import { KpiGrid, KpiCard } from "@/components/atelier/kpi";
import { getStoreData } from "@/lib/shopify/queries";
import { salesInRange, ordersCount, lowStock, firstOfMonthISO, todayISO } from "@/lib/shopify/summary";

export default async function TiendaPage() {
  const company = await getActiveCompany();
  const supabase = await createClient();
  const { data: companies } = await supabase.from("companies").select("id,name").order("name");
  const companyId = company !== "all" ? company : companies && companies.length === 1 ? companies[0].id : "";

  return (
    <div className="space-y-8">
      <PageHeader eyebrow="Operaciones" title="Tienda (Shopify)" />
      {!companyId ? (
        <Aviso>Elige una empresa concreta (selector superior) para ver su tienda Shopify.</Aviso>
      ) : (
        <TiendaBody companyId={companyId} />
      )}
    </div>
  );
}

async function TiendaBody({ companyId }: { companyId: string }) {
  const data = await getStoreData(companyId);

  if ("error" in data) {
    return (
      <Aviso>
        No se pudo conectar con Shopify. Revisa el token en{" "}
        <a className="underline" href="/conexiones">Conexiones</a>.
        <span className="mt-1 block font-mono text-[11px] text-ink-4">{data.error}</span>
      </Aviso>
    );
  }
  if (!data.configured) {
    return (
      <Aviso>
        Conecta tu tienda en <a className="underline" href="/conexiones">Conexiones</a> (proveedor
        Shopify) para ver aquí tus ventas e inventario.
      </Aviso>
    );
  }
  if (!data.access) {
    return <Aviso>Shopify está conectado, pero por ahora solo la propietaria ve estos datos.</Aviso>;
  }

  const now = new Date();
  const monthStart = firstOfMonthISO(now);
  const today = todayISO(now);
  const ventasHoy = salesInRange(data.orders, today, today);
  const ventasMes = salesInRange(data.orders, monthStart);
  const pedidosMes = ordersCount(data.orders, monthStart);
  const activos = data.products.filter((p) => p.status === "ACTIVE").length;
  const bajos = new Set(lowStock(data.products).map((p) => p.title));
  const fmt = (n: number) => `${n.toFixed(2)} ${data.currency === "EUR" ? "€" : data.currency}`;
  const th = "px-3 py-2 text-left font-mono text-[10px] uppercase tracking-[0.06em] text-ink-4";

  return (
    <div className="space-y-8">
      <KpiGrid>
        <KpiCard label="Ventas hoy" value={fmt(ventasHoy)} index="€" />
        <KpiCard label="Ventas del mes" value={fmt(ventasMes)} index="€" />
        <KpiCard label="Pedidos del mes" value={pedidosMes} index="nº" />
        <KpiCard label="Productos activos" value={activos} index="nº" />
      </KpiGrid>

      <section className="space-y-2">
        <h2 className={th.replace("px-3 py-2 ", "")}>Pedidos recientes</h2>
        {data.orders.length === 0 ? (
          <p className="text-sm text-ink-3">Sin pedidos este mes.</p>
        ) : (
          <div className="overflow-x-auto border border-line">
            <table className="w-full text-[13px]">
              <thead>
                <tr className="border-b border-line bg-elevated">
                  <th className={th}>Pedido</th><th className={th}>Cliente</th><th className={th}>Estado</th>
                  <th className={`${th} text-right`}>Total</th>
                </tr>
              </thead>
              <tbody>
                {data.orders.slice(0, 10).map((o) => (
                  <tr key={o.name} className="border-b border-line last:border-0">
                    <td className="px-3 py-1.5 text-ink">{o.name}</td>
                    <td className="px-3 py-1.5 text-ink-3">{o.customer ?? "—"}</td>
                    <td className="px-3 py-1.5 text-ink-3">{o.financialStatus}</td>
                    <td className="px-3 py-1.5 text-right text-ink">{fmt(o.total)}</td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        )}
      </section>

      <section className="space-y-2">
        <h2 className={th.replace("px-3 py-2 ", "")}>Productos</h2>
        <div className="overflow-x-auto border border-line">
          <table className="w-full text-[13px]">
            <thead>
              <tr className="border-b border-line bg-elevated">
                <th className={th}>Producto</th><th className={th}>Estado</th>
                <th className={`${th} text-right`}>Stock</th><th className={`${th} text-right`}>Precio</th>
              </tr>
            </thead>
            <tbody>
              {data.products.map((p) => (
                <tr key={p.title} className="border-b border-line last:border-0">
                  <td className="px-3 py-1.5 text-ink">{p.title}</td>
                  <td className="px-3 py-1.5 text-ink-3">{p.status}</td>
                  <td className={`px-3 py-1.5 text-right ${bajos.has(p.title) ? "text-[var(--danger,#b00)]" : "text-ink"}`}>
                    {p.inventory}{bajos.has(p.title) ? " ⚠" : ""}
                  </td>
                  <td className="px-3 py-1.5 text-right text-ink">{fmt(p.price)}</td>
                </tr>
              ))}
            </tbody>
          </table>
        </div>
      </section>

      <p className="text-[12px] text-ink-4">
        Datos en vivo de Shopify (solo lectura). Tu inventario y ventas manuales siguen en Inventario y Ventas.
```
