# IA Tools (`/ia`) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Construir `/ia`: un asistente IA "analista de negocio" sobre un snapshot agregado de los datos del panel + 3 generadores de texto (email a clienta, descripción de producto, copy social), usando OpenRouter (DeepSeek) server-side.

**Architecture:** Cliente OpenRouter server-only (`fetch`, no-streaming) leído solo desde server actions; la key nunca llega al navegador. Lógica pura de prompts testeable. El chat se ancla a un snapshot agregado (reusa queries existentes de finanzas/analítica/CRM/inventario) — sin tool-calling ni SQL generado. Sin migraciones, sin estado persistido.

**Tech Stack:** Next.js 16 (App Router, server actions), Supabase (lectura con RLS), OpenRouter (API compatible OpenAI vía fetch), shadcn/ui, Atelier, sonner, Vitest.

**Reglas del proyecto:** NO tocar `src/components/app-shell/sidebar.tsx` (la entrada `/ia` la enciende la terminal de diseño). Namespace `ia`. SIN migraciones. Integrar por `main`. Rama: `feat/ia`.

**Variables de entorno** (server-only; añadir a `.env.local` y a Vercel): `OPENROUTER_API_KEY` (ya presente), `OPENROUTER_MODEL_PRO` (default `deepseek/deepseek-r1`), `OPENROUTER_MODEL_FAST` (default `deepseek/deepseek-chat`).

```ts
export type ChatMessage = { role: "system" | "user" | "assistant"; content: string };

export const MODEL_PRO = process.env.OPENROUTER_MODEL_PRO ?? "deepseek/deepseek-r1";
export const MODEL_FAST = process.env.OPENROUTER_MODEL_FAST ?? "deepseek/deepseek-chat";

export async function chatComplete(
  messages: ChatMessage[],
  model: string,
  maxTokens = 1000,
): Promise<{ text: string } | { error: string }> {
  const key = process.env.OPENROUTER_API_KEY;
  if (!key) return { error: "Falta OPENROUTER_API_KEY en el entorno." };
  try {
    const res = await fetch("https://openrouter.ai/api/v1/chat/completions", {
      method: "POST",
      headers: {
        Authorization: `Bearer ${key}`,
        "Content-Type": "application/json",
      },
      body: JSON.stringify({ model, messages, max_tokens: maxTokens }),
    });
    if (!res.ok) {
      const body = await res.text();
      return { error: `OpenRouter ${res.status}: ${body.slice(0, 200)}` };
    }
    const data = await res.json();
    const text = data?.choices?.[0]?.message?.content;
    if (typeof text !== "string" || !text.trim()) return { error: "Respuesta vacía del modelo." };
    return { text: text.trim() };
  } catch (e) {
    return { error: e instanceof Error ? e.message : "Error de red llamando a OpenRouter." };
  }
}
```

- [ ] **Step 1: Crear `src/lib/ia/queries.ts`** con exactamente:

```ts
import { financeOverview } from "@/lib/finanzas/queries";
import { salesByChannel } from "@/lib/analitica/queries";
import { crmSummary } from "@/lib/crm/queries";
import { inventorySummary } from "@/lib/inventory/queries";

export async function buildBusinessSnapshot(companyFilter: string | "all"): Promise<string> {
  const year = new Date().getFullYear();
  const [fin, channels, crm, inv] = await Promise.all([
    financeOverview(companyFilter, year),
    salesByChannel(companyFilter, year),
    crmSummary(companyFilter),
    inventorySummary(companyFilter),
  ]);
  const eur = (n: number) => `${Math.round(n).toLocaleString("es-ES")} €`;
  const channelsTxt = channels.length
    ? channels.map((c) => `${c.channel}: ${eur(c.total)}`).join(", ")
    : "sin ventas registradas";
  return [
    `Año: ${year}.`,
    `Finanzas: ingresos ${eur(fin.totals.ingresos)}, gastos ${eur(fin.totals.gastos)}, ` +
      `resultado ${eur(fin.totals.resultado)} (margen ${fin.totals.margenPct.toFixed(1)}%).`,
    `Ventas por canal: ${channelsTxt}.`,
    `Clientas: ${crm.totalCustomers} en total (${crm.activeCount} activas, ${crm.leadCount} leads).`,
    `Inventario: ${inv.totalProducts} productos, valor de stock ${eur(inv.stockValue)}, ` +
      `${inv.lowStock.length} con stock bajo.`,
  ].join("\n");
}
```

```ts
"use server";
import { getActiveCompany } from "@/lib/active-company";
import { buildBusinessSnapshot } from "./queries";
import { chatComplete, MODEL_PRO, MODEL_FAST } from "./openrouter";
import {
  buildAnalystMessages, buildEmailMessages, buildProductDescriptionMessages, buildSocialCopyMessages,
} from "./prompts";

export async function askAnalyst(question: string) {
  const q = question.trim();
  if (!q) return { error: "Escribe una pregunta." };
  const company = await getActiveCompany();
  const snapshot = await buildBusinessSnapshot(company);
  return chatComplete(buildAnalystMessages(snapshot, q), MODEL_PRO, 1200);
}

export async function generateCustomerEmail(formData: FormData) {
  const customerName = String(formData.get("customerName") ?? "").trim();
  const goal = String(formData.get("goal") ?? "").trim();
  const notes = String(formData.get("notes") ?? "").trim() || undefined;
  if (!customerName || !goal) return { error: "Faltan datos (clienta u objetivo)." };
  return chatComplete(buildEmailMessages({ customerName, goal, notes }), MODEL_FAST, 600);
}

export async function generateProductDescription(formData: FormData) {
  const name = String(formData.get("name") ?? "").trim();
  const category = String(formData.get("category") ?? "").trim() || null;
  const price = Number(formData.get("price") ?? 0);
  if (!name) return { error: "Falta el producto." };
  return chatComplete(buildProductDescriptionMessages({ name, category, price }), MODEL_FAST, 400);
}

export async function generateSocialCopy(formData: FormData) {
  const topic = String(formData.get("topic") ?? "").trim();
  const platform = String(formData.get("platform") ?? "instagram").trim();
  const tone = String(formData.get("tone") ?? "elegante").trim();
  if (!topic) return { error: "Escribe un tema o producto." };
  return chatComplete(buildSocialCopyMessages({ topic, platform, tone }), MODEL_FAST, 400);
}
```

```tsx
export function GeneratedOutput({ text }: { text: string }) {
```

```tsx
"use client";
import { useState } from "react";
import { toast } from "sonner";
import { Button } from "@/components/ui/button";
import { Textarea } from "@/components/ui/textarea";
import { askAnalyst } from "@/lib/ia/actions";

type Msg = { role: "user" | "assistant"; content: string };
const EXAMPLES = [
  "¿Qué clientas debería reactivar?",
  "¿Qué producto empujar este mes?",
  "¿Dónde estoy perdiendo margen?",
];

export function AnalystChat() {
  const [messages, setMessages] = useState<Msg[]>([]);
  const [input, setInput] = useState("");
  const [loading, setLoading] = useState(false);

  async function send(question: string) {
    const q = question.trim();
    if (!q || loading) return;
    setMessages((m) => [...m, { role: "user", content: q }]);
    setInput("");
    setLoading(true);
    const res = await askAnalyst(q);
    setLoading(false);
    if ("error" in res) { toast.error(res.error); return; }
    setMessages((m) => [...m, { role: "assistant", content: res.text }]);
  }

  return (
    <div className="space-y-4">
      {messages.length === 0 && (
        <div className="flex flex-wrap gap-2">
          {EXAMPLES.map((ex) => (
            <button key={ex} type="button" onClick={() => send(ex)}
              className="rounded-full border border-line px-3 py-1 text-sm text-ink-3 hover:bg-paper">
              {ex}
            </button>
          ))}
        </div>
      )}
      <div className="space-y-3">
        {messages.map((m, i) => (
          <div key={i} className={m.role === "user" ? "text-right" : ""}>
            <div className={`inline-block max-w-[85%] whitespace-pre-wrap rounded-md p-3 text-left text-sm ${m.role === "user" ? "bg-brand-soft text-ink" : "border border-line bg-elevated text-ink"}`}>
              {m.content}
            </div>
          </div>
        ))}
        {loading && <p className="text-sm text-ink-3">Analizando…</p>}
      </div>
      <form onSubmit={(e) => { e.preventDefault(); send(input); }} className="flex gap-2">
        <Textarea value={input} onChange={(e) => setInput(e.target.value)}
          placeholder="Pregunta sobre tus datos…" rows={2} className="flex-1" />
        <Button type="submit" disabled={loading}>Enviar</Button>
      </form>
    </div>
  );
}
```

```tsx
import Link from "next/link";
import { getActiveCompany } from "@/lib/active-company";
import { listCustomers } from "@/lib/crm/queries";
import { listProducts } from "@/lib/inventory/queries";
import { GeneratorEmail } from "@/components/ia/generator-email";
import { GeneratorProduct } from "@/components/ia/generator-product";
import { GeneratorSocial } from "@/components/ia/generator-social";
import { Button } from "@/components/ui/button";

export default async function GeneradoresPage() {
  const company = await getActiveCompany();
  const [customers, products] = await Promise.all([listCustomers(company), listProducts(company)]);
  return (
    <div className="space-y-6">
      <div className="flex items-center justify-between gap-3">
        <h1 className="text-2xl font-semibold">Generadores IA</h1>
        <Button asChild variant="outline"><Link href="/ia">Asistente</Link></Button>
      </div>
      <div className="grid gap-4 lg:grid-cols-3">
        <GeneratorEmail customers={customers.map((c) => ({ id: c.id, name: c.name }))} />
        <GeneratorProduct products={products.map((p) => ({ id: p.id, name: p.name, category: p.category?.name ?? null, price: p.price }))} />
        <GeneratorSocial />
      </div>
    </div>
  );
}
```
