> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Construir el módulo Tareas de equipo: un tablero por columnas (Pendiente/En curso/Hecha) con tareas (área, prioridad, fecha límite, asignado), aislado por empresa, Atelier-native.

**Architecture:** Una tabla Supabase `tasks` con RLS por `company_id`. Lógica pura testeada (labels + `isOverdue`). Lecturas en `queries.ts`, mutaciones en `actions.ts` (server actions con toasts). Tablero cliente que reparte las tareas en 3 columnas y mueve estados con botones (server action `setTaskStatus`). UI con tokens/primitivas Atelier (`StatusPill`, toasts). Deploy = push a `main`.

**Tech Stack:** Next.js 15 (App Router, TS), Supabase (`@supabase/ssr`), shadcn/ui + Atelier, sonner, Vitest, Vercel.

**Spec:** `docs/superpowers/specs/2026-05-26-tareas-design.md`

```typescript
export type TaskStatus = "pendiente" | "en_curso" | "hecha";
export type TaskPriority = "baja" | "media" | "alta";
export type TaskArea = "inventario" | "comercial" | "ventas" | "finanzas" | "general";

const STATUS_LABELS: Record<TaskStatus, string> = {
  pendiente: "Pendiente",
  en_curso: "En curso",
  hecha: "Hecha",
};
const PRIORITY_LABELS: Record<TaskPriority, string> = {
  baja: "Baja",
  media: "Media",
  alta: "Alta",
};
const AREA_LABELS: Record<TaskArea, string> = {
  inventario: "Inventario",
  comercial: "Comercial",
  ventas: "Ventas",
  finanzas: "Finanzas",
  general: "General",
};

export function taskStatusLabel(s: TaskStatus): string {
  return STATUS_LABELS[s];
}
export function priorityLabel(p: TaskPriority): string {
  return PRIORITY_LABELS[p];
}
export function areaLabel(a: TaskArea): string {
  return AREA_LABELS[a];
}

export function isOverdue(
  dueDate: string | null,
  status: TaskStatus,
  today: string = new Date().toISOString().slice(0, 10)
): boolean {
  if (!dueDate) return false;
  if (status === "hecha") return false;
  return dueDate < today;
}
```

```typescript
import { createClient } from "@/lib/supabase/server";
import { isOverdue, type TaskStatus, type TaskPriority, type TaskArea } from "./task";

export type TaskRow = {
  id: string;
  title: string;
  description: string | null;
  status: TaskStatus;
  priority: TaskPriority;
  area: TaskArea;
  due_date: string | null;
  assignee: { full_name: string | null } | null;
};

export async function listTasks(companyFilter: string | "all"): Promise<TaskRow[]> {
  const supabase = await createClient();
  let q = supabase
    .from("tasks")
    .select("id,title,description,status,priority,area,due_date,assignee:profiles!tasks_assignee_id_fkey(full_name)")
    .order("due_date", { ascending: true, nullsFirst: false })
    .order("created_at", { ascending: false });
  if (companyFilter !== "all") q = q.eq("company_id", companyFilter);
  const { data, error } = await q;
  if (error) throw error;
  return (data ?? []).map((r) => ({
    id: r.id,
    title: r.title,
    description: r.description,
    status: r.status as TaskStatus,
    priority: r.priority as TaskPriority,
    area: r.area as TaskArea,
    due_date: r.due_date,
    assignee: r.assignee as { full_name: string | null } | null,
  }));
}

export async function getTask(id: string) {
  const supabase = await createClient();
  const { data, error } = await supabase.from("tasks").select("*").eq("id", id).single();
  if (error) throw error;
  return data;
}

export async function tasksSummary(companyFilter: string | "all") {
  const tasks = await listTasks(companyFilter);
  const pendientes = tasks.filter((t) => t.status === "pendiente").length;
  const enCurso = tasks.filter((t) => t.status === "en_curso").length;
  const vencidas = tasks.filter((t) => isOverdue(t.due_date, t.status)).length;
  return { pendientes, enCurso, vencidas };
}

export async function profilesForAssignee() {
  const supabase = await createClient();
  const { data, error } = await supabase.from("profiles").select("id,full_name").order("full_name");
  if (error) throw error;
  return (data ?? []) as { id: string; full_name: string | null }[];
}
```
Nota sobre el join: la FK `assignee_id` requiere desambiguar con `profiles!tasks_assignee_id_fkey` porque `tasks` también tiene `created_by` hacia `profiles`. Si el nombre exacto de la constraint difiere, ajustar al que muestre `list_tables`/db.ts (formato `tasks_assignee_id_fkey`).

```typescript
"use server";
import { createClient } from "@/lib/supabase/server";
import { revalidatePath } from "next/cache";
import type { TaskStatus } from "./task";

type StatusV = "pendiente" | "en_curso" | "hecha";
type PriorityV = "baja" | "media" | "alta";
type AreaV = "inventario" | "comercial" | "ventas" | "finanzas" | "general";

function assignee(formData: FormData): string | null {
  const v = formData.get("assignee_id");
  return v && v !== "none" ? String(v) : null;
}

export async function createTask(formData: FormData) {
  const supabase = await createClient();
  const { data: { user } } = await supabase.auth.getUser();
  const { error } = await supabase.from("tasks").insert({
    company_id: String(formData.get("company_id")),
    title: String(formData.get("title")),
    description: (formData.get("description") as string) || null,
    priority: formData.get("priority") as PriorityV,
    area: formData.get("area") as AreaV,
    assignee_id: assignee(formData),
    due_date: (formData.get("due_date") as string) || null,
    created_by: user?.id ?? null,
  });
  if (error) return { error: error.message };
  revalidatePath("/tareas");
  return { ok: true };
}

export async function updateTask(id: string, formData: FormData) {
  const supabase = await createClient();
  const { error } = await supabase.from("tasks").update({
    title: String(formData.get("title")),
    description: (formData.get("description") as string) || null,
    priority: formData.get("priority") as PriorityV,
    area: formData.get("area") as AreaV,
    assignee_id: assignee(formData),
    due_date: (formData.get("due_date") as string) || null,
    updated_at: new Date().toISOString(),
  }).eq("id", id);
  if (error) return { error: error.message };
  revalidatePath("/tareas");
  return { ok: true };
}

export async function setTaskStatus(id: string, status: TaskStatus) {
  const supabase = await createClient();
  const { error } = await supabase.from("tasks")
    .update({ status, updated_at: new Date().toISOString() }).eq("id", id);
  if (error) return { error: error.message };
  revalidatePath("/tareas");
  return { ok: true };
}

export async function deleteTask(id: string) {
  const supabase = await createClient();
  const { error } = await supabase.from("tasks").delete().eq("id", id);
  if (error) return { error: error.message };
  revalidatePath("/tareas");
  return { ok: true };
}
```
Nota: el `status` no se edita en el formulario (se mueve en el tablero), por eso `updateTask` no lo toca.

```tsx
"use client";
import { useState } from "react";
import { useRouter } from "next/navigation";
import Link from "next/link";
import { toast } from "sonner";
import { Button } from "@/components/ui/button";
import { StatusPill } from "@/components/atelier/status-pill";
import { areaLabel, priorityLabel, isOverdue, type TaskStatus, type TaskPriority } from "@/lib/tareas/task";
import { setTaskStatus, deleteTask } from "@/lib/tareas/actions";
import type { TaskRow } from "@/lib/tareas/queries";

const PRIORITY_TONE: Record<TaskPriority, "bad" | "warn" | "neutral"> = {
  alta: "bad",
  media: "warn",
  baja: "neutral",
};

const MOVES: Record<TaskStatus, { status: TaskStatus; label: string }[]> = {
  pendiente: [{ status: "en_curso", label: "→ En curso" }],
  en_curso: [{ status: "pendiente", label: "← Pendiente" }, { status: "hecha", label: "→ Hecha" }],
  hecha: [{ status: "en_curso", label: "← En curso" }],
};

export function TaskCard({ task }: { task: TaskRow }) {
  const router = useRouter();
  const [pending, setPending] = useState(false);
  const overdue = isOverdue(task.due_date, task.status);

  async function move(status: TaskStatus) {
    if (pending) return;
    setPending(true);
    const res = await setTaskStatus(task.id, status);
    setPending(false);
    if (res?.error) { toast.error(res.error); return; }
    router.refresh();
  }
  async function onDelete() {
    if (pending) return;
    if (!confirm("¿Borrar esta tarea?")) return;
    setPending(true);
    const res = await deleteTask(task.id);
    setPending(false);
    if (res?.error) { toast.error(res.error); return; }
    toast.success("Tarea borrada");
    router.refresh();
  }

  return (
    <div className="space-y-2 rounded-md border border-line bg-elevated p-3">
      <div className="flex items-start justify-between gap-2">
        <p className="text-sm font-medium text-ink">{task.title}</p>
        <StatusPill tone={PRIORITY_TONE[task.priority]}>{priorityLabel(task.priority)}</StatusPill>
      </div>
      <div className="flex flex-wrap items-center gap-x-3 gap-y-1 font-mono text-[10.5px] uppercase tracking-[0.04em] text-ink-4">
        <span>{areaLabel(task.area)}</span>
        {task.assignee?.full_name && <span>· {task.assignee.full_name}</span>}
        {task.due_date && (
          <span className={overdue ? "text-bad" : undefined}>
            · {new Date(task.due_date).toLocaleDateString("es-ES", { timeZone: "UTC" })}{overdue ? " (vencida)" : ""}
          </span>
        )}
      </div>
      <div className="flex flex-wrap items-center gap-2 pt-1">
        {MOVES[task.status].map((m) => (
          <Button key={m.status} variant="outline" size="sm" disabled={pending} onClick={() => move(m.status)}>{m.label}</Button>
        ))}
        <Button asChild variant="ghost" size="sm"><Link href={`/tareas/${task.id}/editar`}>Editar</Link></Button>
        <Button variant="ghost" size="sm" disabled={pending} onClick={onDelete}>Borrar</Button>
      </div>
    </div>
  );
}
```

```tsx
"use client";
import { useState } from "react";
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@/components/ui/select";
import { TaskCard } from "./task-card";
import { taskStatusLabel, areaLabel, type TaskStatus, type TaskArea } from "@/lib/tareas/task";

const COLUMNS: TaskStatus[] = ["pendiente", "en_curso", "hecha"];

export function TaskBoard({ tasks }: { tasks: TaskRow[] }) {
  const [area, setArea] = useState("all");
  const filtered = area === "all" ? tasks : tasks.filter((t) => t.area === area);

  return (
    <div className="space-y-4">
      <div className="flex flex-wrap gap-3">
        <Select value={area} onValueChange={setArea}>
          <SelectTrigger className="w-48" aria-label="Filtrar por área"><SelectValue placeholder="Área" /></SelectTrigger>

      </div>

        {COLUMNS.map((col) => {
          const items = filtered.filter((t) => t.status === col);
          return (
            <div key={col} className="space-y-3">
              <div className="flex items-center justify-between border-b border-line pb-2">

              </div>
              {items.length === 0 ? (
                <p className="text-sm text-ink-4">—</p>
              ) : (
                items.map((t) => <TaskCard key={t.id} task={t} />)
              )}
            </div>
          );
        })}
      </div>
    </div>
  );
}
```

```tsx
const AREAS = [["inventario","Inventario"],["comercial","Comercial"],["ventas","Ventas"],["finanzas","Finanzas"],["general","General"]] as const;
const PRIORITIES = [["baja","Baja"],["media","Media"],["alta","Alta"]] as const;

export function TaskForm({ companyId, profiles, defaults, action }: {
  companyId: string;
  profiles: Profile[];
  defaults?: Defaults;
  action: (fd: FormData) => Promise<{ error?: string; ok?: boolean }>;
}) {
  const router = useRouter();
  async function onSubmit(fd: FormData) {
    const res = await action(fd);
    if (res?.error) { toast.error(res.error); return; }
    toast.success("Tarea guardada");
    router.push("/tareas");
  }
  return (
    <form action={onSubmit} className="max-w-lg space-y-4">
      <input type="hidden" name="company_id" value={companyId} />
      <div className="space-y-2">
        <Label htmlFor="title">Título</Label>
        <Input id="title" name="title" defaultValue={defaults?.title} required />
      </div>
      <div className="space-y-2">
        <Label htmlFor="description">Descripción</Label>
        <Textarea id="description" name="description" defaultValue={defaults?.description} />
      </div>
      <div className="grid grid-cols-2 gap-4">
        <div className="space-y-2">
          <Label>Área</Label>
          <Select name="area" defaultValue={defaults?.area ?? "general"}>
            <SelectTrigger aria-label="Área"><SelectValue /></SelectTrigger>
            <SelectContent>{AREAS.map(([v, l]) => <SelectItem key={v} value={v}>{l}</SelectItem>)}</SelectContent>
          </Select>
        </div>
        <div className="space-y-2">
          <Label>Prioridad</Label>
          <Select name="priority" defaultValue={defaults?.priority ?? "media"}>
            <SelectTrigger aria-label="Prioridad"><SelectValue /></SelectTrigger>
            <SelectContent>{PRIORITIES.map(([v, l]) => <SelectItem key={v} value={v}>{l}</SelectItem>)}</SelectContent>
          </Select>
        </div>
        <div className="space-y-2">
          <Label>Asignado</Label>
          <Select name="assignee_id" defaultValue={defaults?.assignee_id ?? "none"}>
            <SelectTrigger aria-label="Asignado"><SelectValue /></SelectTrigger>
            <SelectContent>
              <SelectItem value="none">Sin asignar</SelectItem>
              {profiles.map((p) => <SelectItem key={p.id} value={p.id}>{p.full_name ?? "Usuario"}</SelectItem>)}
            </SelectContent>
          </Select>
        </div>
        <div className="space-y-2">
          <Label htmlFor="due_date">Fecha límite</Label>
          <Input id="due_date" name="due_date" type="date" defaultValue={defaults?.due_date} />
        </div>
      </div>
      <Button type="submit">Guardar</Button>
    </form>
  );
}
```

```tsx
import { getTask, profilesForAssignee } from "@/lib/tareas/queries";
import { updateTask } from "@/lib/tareas/actions";
import { TaskForm } from "@/components/tareas/task-form";

export default async function EditarTareaPage({ params }: { params: Promise<{ id: string }> }) {
  const { id } = await params;
  const [task, profiles] = await Promise.all([getTask(id), profilesForAssignee()]);
  const defaults = {
    title: task.title as string,
    description: (task.description as string | null) ?? undefined,
    priority: task.priority as string,
    area: task.area as string,
    assignee_id: (task.assignee_id as string | null) ?? undefined,
    due_date: (task.due_date as string | null) ?? undefined,
  };
  return (
    <div className="space-y-6">
      <h1 className="text-2xl font-semibold">Editar tarea</h1>
      <TaskForm companyId={task.company_id as string} profiles={profiles} defaults={defaults} action={updateTask.bind(null, id)} />
    </div>
  );
}
```
