function round2(n: number): number {
  return Math.round((n + Number.EPSILON) * 100) / 100;
}

export function financeSummary(input: {
  salesRevenue: number; manualIncome: number; cogs: number; expenses: number;
}): { ingresos: number; gastos: number; resultado: number; margen: number; margenPct: number } {
  const ingresos = round2(input.salesRevenue + input.manualIncome);
  const gastos = round2(input.expenses);
  const resultado = round2(ingresos - gastos);
  const margen = round2(ingresos - input.cogs - gastos);
  const margenPct = ingresos > 0 ? round2((margen / ingresos) * 100) : 0;
  return { ingresos, gastos, resultado, margen, margenPct };
}
```

import { createClient } from "@/lib/supabase/server";
import { financeSummary, type ExpenseCategory } from "./finance";

export type ExpenseRow = {
  id: string;
  category: ExpenseCategory;
  supplier: string | null;
  amount: number;
  expense_date: string;
  recurring: boolean;
  note: string | null;
};

export type IncomeRow = {
  id: string;
  concept: string;
  amount: number;
  income_date: string;
  note: string | null;
};

function yearRange(year: number) {
  return { start: `${year}-01-01`, end: `${year}-12-31` };
}

export async function listExpenses(companyFilter: string | "all", year: number): Promise<ExpenseRow[]> {
  const supabase = await createClient();
  const { start, end } = yearRange(year);
  let q = supabase
    .from("expenses")
    .select("id,category,supplier,amount,expense_date,recurring,note")
    .gte("expense_date", start).lte("expense_date", end)
    .order("expense_date", { ascending: false });
  if (companyFilter !== "all") q = q.eq("company_id", companyFilter);
  const { data, error } = await q;
  if (error) throw error;
  return (data ?? []).map((r) => ({ ...r, category: r.category as ExpenseCategory })) as ExpenseRow[];
}

export async function listIncomes(companyFilter: string | "all", year: number): Promise<IncomeRow[]> {
  const supabase = await createClient();
  const { start, end } = yearRange(year);
  let q = supabase
    .from("incomes")
    .select("id,concept,amount,income_date,note")
    .gte("income_date", start).lte("income_date", end)
    .order("income_date", { ascending: false });
  if (companyFilter !== "all") q = q.eq("company_id", companyFilter);
  const { data, error } = await q;
  if (error) throw error;
  return (data ?? []) as IncomeRow[];
}

export async function getExpense(id: string) {
  const supabase = await createClient();
  const { data, error } = await supabase.from("expenses").select("*").eq("id", id).single();
  if (error) throw error;
  return data;
}

export type MonthFinance = {
  month: number; // 1-12
  salesRevenue: number;
  cogs: number;
  manualIncome: number;
  ingresos: number;
  gastos: number;
  resultado: number;
  margen: number;
};
export type FinanceOverview = {
  year: number;
  months: MonthFinance[];
  totals: { ingresos: number; gastos: number; resultado: number; margen: number; margenPct: number; salesRevenue: number; cogs: number; manualIncome: number };
};

export async function financeOverview(companyFilter: string | "all", year: number): Promise<FinanceOverview> {
  const supabase = await createClient();
  const { start, end } = yearRange(year);
  const monthOf = (d: string) => Number(d.slice(5, 7)); // "YYYY-MM-DD" -> 1..12

  // 1) Ventas pagadas del año
  let salesQ = supabase
    .from("sales")
    .select("id,total,sale_date")
    .eq("status", "pagada")
    .gte("sale_date", start).lte("sale_date", end);
  if (companyFilter !== "all") salesQ = salesQ.eq("company_id", companyFilter);
  const { data: sales, error: sErr } = await salesQ;
  if (sErr) throw sErr;
  const saleMonth = new Map<string, number>();
  for (const s of sales ?? []) saleMonth.set(s.id, monthOf(s.sale_date));

  // 2) COGS: líneas de esas ventas × coste del producto
  const cogsByMonth = new Array(13).fill(0); // index 1..12
  const saleIds = (sales ?? []).map((s) => s.id);
  if (saleIds.length > 0) {
    const { data: items, error: iErr } = await supabase
      .from("sale_items")
      .select("sale_id,quantity,product:products(cost)")
      .in("sale_id", saleIds);
    if (iErr) throw iErr;
    for (const it of items ?? []) {
      const m = saleMonth.get(it.sale_id as string);
      if (!m) continue;
      const cost = ((it.product as { cost: number } | null)?.cost) ?? 0;
      cogsByMonth[m] += Number(it.quantity) * Number(cost);
    }
  }

  // 3) Ingresos de ventas por mes
  const salesByMonth = new Array(13).fill(0);
  for (const s of sales ?? []) salesByMonth[monthOf(s.sale_date)] += Number(s.total);

  // 4) Ingresos manuales y gastos por mes
  const incomes = await listIncomes(companyFilter, year);
  const expenses = await listExpenses(companyFilter, year);
  const manualByMonth = new Array(13).fill(0);
  for (const i of incomes) manualByMonth[monthOf(i.income_date)] += Number(i.amount);
  const expensesByMonth = new Array(13).fill(0);
  for (const e of expenses) expensesByMonth[monthOf(e.expense_date)] += Number(e.amount);

  const months: MonthFinance[] = [];
  let tSales = 0, tCogs = 0, tManual = 0, tExpenses = 0;
  for (let m = 1; m <= 12; m++) {
    const sum = financeSummary({
      salesRevenue: salesByMonth[m], manualIncome: manualByMonth[m],
      cogs: cogsByMonth[m], expenses: expensesByMonth[m],
    });
    months.push({
      month: m,
      salesRevenue: salesByMonth[m], cogs: cogsByMonth[m], manualIncome: manualByMonth[m],
      ingresos: sum.ingresos, gastos: sum.gastos, resultado: sum.resultado, margen: sum.margen,
    });
    tSales += salesByMonth[m]; tCogs += cogsByMonth[m]; tManual += manualByMonth[m]; tExpenses += expensesByMonth[m];
  }
  const totalsSum = financeSummary({ salesRevenue: tSales, manualIncome: tManual, cogs: tCogs, expenses: tExpenses });
  return {
    year, months,
    totals: { ...totalsSum, salesRevenue: tSales, cogs: tCogs, manualIncome: tManual },
  };
}
```

"use server";
import { createClient } from "@/lib/supabase/server";
import { revalidatePath } from "next/cache";

type ExpenseCategoryValue =
  "mercancia" | "marketing" | "personal" | "alquiler" | "suministros" | "logistica" | "comisiones" | "impuestos" | "otros";

export async function createExpense(formData: FormData) {
  const supabase = await createClient();
  const { data: { user } } = await supabase.auth.getUser();
  const { error } = await supabase.from("expenses").insert({
    company_id: String(formData.get("company_id")),
    category: formData.get("category") as ExpenseCategoryValue,
    supplier: (formData.get("supplier") as string) || null,
    amount: Number(formData.get("amount") || 0),
    expense_date: String(formData.get("expense_date")),
    recurring: formData.get("recurring") === "on" || formData.get("recurring") === "true",
    note: (formData.get("note") as string) || null,
    created_by: user?.id ?? null,
  });
  if (error) return { error: error.message };
  revalidatePath("/finanzas/gastos");
  revalidatePath("/finanzas");
  return { ok: true };
}

Create `src/components/finanzas/expense-table.tsx`:
```tsx
"use client";
import { useState } from "react";
import { useRouter } from "next/navigation";
import Link from "next/link";
import { toast } from "sonner";
import { Table, TableBody, TableCell, TableHead, TableHeader, TableRow } from "@/components/ui/table";
import { Button } from "@/components/ui/button";
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@/components/ui/select";
import { Money } from "@/components/atelier/money";
import { expenseCategoryLabel, type ExpenseCategory } from "@/lib/finanzas/finance";
import { deleteExpense } from "@/lib/finanzas/actions";
import type { ExpenseRow } from "@/lib/finanzas/queries";

export function ExpenseTable({ expenses }: { expenses: ExpenseRow[] }) {
  const router = useRouter();
  const [category, setCategory] = useState("all");
  const [recurring, setRecurring] = useState("all");

  const filtered = expenses.filter((e) => {
    const matchesCat = category === "all" || e.category === category;
    const matchesRec = recurring === "all" || (recurring === "yes" ? e.recurring : !e.recurring);
    return matchesCat && matchesRec;
  });

  async function onDelete(id: string) {
    if (!confirm("¿Borrar este gasto?")) return;
    const res = await deleteExpense(id);
    if (res?.error) { toast.error(res.error); return; }
    toast.success("Gasto borrado");
    router.refresh();
  }

  return (
    <div className="space-y-4">
      <div className="flex flex-wrap gap-3">
        <Select value={category} onValueChange={setCategory}>
          <SelectTrigger className="w-44" aria-label="Filtrar por categoría"><SelectValue placeholder="Categoría" /></SelectTrigger>
          <SelectContent>
            <SelectItem value="all">Todas las categorías</SelectItem>
            {(["mercancia","marketing","personal","alquiler","suministros","logistica","comisiones","impuestos","otros"] as ExpenseCategory[]).map((c) =>
              <SelectItem key={c} value={c}>{expenseCategoryLabel(c)}</SelectItem>)}
          </SelectContent>
        </Select>
        <Select value={recurring} onValueChange={setRecurring}>
          <SelectTrigger className="w-40" aria-label="Filtrar por recurrencia"><SelectValue placeholder="Recurrencia" /></SelectTrigger>
          <SelectContent>
            <SelectItem value="all">Todos</SelectItem>
            <SelectItem value="yes">Recurrentes</SelectItem>
            <SelectItem value="no">Puntuales</SelectItem>
          </SelectContent>
        </Select>
      </div>

      {filtered.length === 0 ? (
        <p className="text-sm text-ink-3">No hay gastos que coincidan.</p>
      ) : (
        <Table>
          <TableHeader>
            <TableRow>
              <TableHead>Fecha</TableHead><TableHead>Categoría</TableHead><TableHead>Proveedor</TableHead>
              <TableHead className="text-right">Importe</TableHead><TableHead></TableHead>
            </TableRow>
          </TableHeader>
          <TableBody>
            {filtered.map((e) => (
              <TableRow key={e.id}>
                <TableCell>{new Date(e.expense_date).toLocaleDateString("es-ES", { timeZone: "UTC" })}</TableCell>
                <TableCell>{expenseCategoryLabel(e.category)}{e.recurring && <span className="ml-2 font-mono text-[10px] uppercase text-ink-4">recurrente</span>}</TableCell>
                <TableCell>{e.supplier ?? "—"}</TableCell>
                <TableCell className="text-right"><Money value={e.amount} /></TableCell>
                <TableCell className="text-right">
                  <Button asChild variant="outline" size="sm"><Link href={`/finanzas/gastos/${e.id}/editar`}>Editar</Link></Button>
                  <Button variant="ghost" size="sm" onClick={() => onDelete(e.id)}>Borrar</Button>
                </TableCell>
              </TableRow>
            ))}
          </TableBody>
        </Table>
      )}
    </div>
  );
}
```

import { useRouter, usePathname, useSearchParams } from "next/navigation";
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@/components/ui/select";

export function YearSelect({ year }: { year: number }) {
  const router = useRouter();
  const pathname = usePathname();
  const params = useSearchParams();
  const now = new Date().getFullYear();
  const years = [now, now - 1, now - 2, now - 3];
  function onChange(value: string) {
    const sp = new URLSearchParams(params.toString());
    sp.set("year", value);
    router.push(`${pathname}?${sp.toString()}`);
  }
  return (
    <Select value={String(year)} onValueChange={onChange}>
      <SelectTrigger className="w-28" aria-label="Año"><SelectValue /></SelectTrigger>
      <SelectContent>
        {years.map((y) => <SelectItem key={y} value={String(y)}>{y}</SelectItem>)}
      </SelectContent>
    </Select>
  );
}
