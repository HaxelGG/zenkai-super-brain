# Módulo CRM — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Construir el módulo CRM (clientes + historial de interacciones, aislado por empresa) sobre los cimientos ya existentes del Panel de Control, replicando 1:1 el patrón del módulo Inventario.

**Architecture:** Dos tablas Supabase nuevas (`customers`, `interactions`) con RLS por `company_id` usando el helper `accessible_company_ids()` ya existente. Lógica pura testeada (parseo de etiquetas, labels) en `src/lib/crm/customer.ts`; lecturas en `queries.ts`; mutaciones en `actions.ts` (server actions). UI con shadcn/ui en `src/app/(app)/crm/`. Las queries filtran por la empresa activa del selector (cookie `active_company`) igual que Inventario. Nada de los cimientos se reconstruye.

**Tech Stack:** Next.js 15 (App Router, TypeScript), Supabase (`@supabase/ssr`), shadcn/ui + Tailwind, Vitest, Vercel. Todo ya instalado y configurado por el plan de cimientos.

**Spec:** `docs/superpowers/specs/2026-05-25-crm-design.md`

create type public.customer_type    as enum ('particular', 'tienda', 'mayorista');
create type public.customer_status  as enum ('lead', 'active', 'inactive');
create type public.interaction_type as enum ('llamada', 'email', 'visita', 'whatsapp', 'otro');

-- Clientes (aislados por empresa, mismo patrón que products)
create table public.customers (
  id uuid primary key default gen_random_uuid(),
  company_id uuid not null references public.companies(id) on delete cascade,
  name text not null,
  type public.customer_type not null default 'particular',
  status public.customer_status not null default 'lead',
  email text,
  phone text,
  tags text[] not null default '{}',
  notes text,
  created_by uuid references public.profiles(id),
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

-- Interacciones (historial por cliente, mismo patrón que stock_movements)
create table public.interactions (
  id uuid primary key default gen_random_uuid(),
  customer_id uuid not null references public.customers(id) on delete cascade,
  type public.interaction_type not null,
  summary text not null,
  occurred_at timestamptz not null default now(),
  created_by uuid references public.profiles(id),
  created_at timestamptz not null default now()
);

create index on public.customers (company_id);
create index on public.interactions (customer_id);

export type CustomerStatus = "lead" | "active" | "inactive";
export type CustomerType = "particular" | "tienda" | "mayorista";
export type InteractionType = "llamada" | "email" | "visita" | "whatsapp" | "otro";

export function parseTags(input: string): string[] {
  const result: string[] = [];
  const seen = new Set<string>();
  for (const raw of input.split(",")) {
    const tag = raw.trim();
    if (!tag) continue;
    const key = tag.toLowerCase();
    if (seen.has(key)) continue;
    seen.add(key);
    result.push(tag);
  }
  return result;
}

const STATUS_LABELS: Record<CustomerStatus, string> = {
  lead: "Lead",
  active: "Activo",
  inactive: "Inactivo",
};

const TYPE_LABELS: Record<CustomerType, string> = {
  particular: "Particular",
  tienda: "Tienda",
  mayorista: "Mayorista",
};

export function statusLabel(status: CustomerStatus): string {
  return STATUS_LABELS[status];
}

export function typeLabel(type: CustomerType): string {
  return TYPE_LABELS[type];
}

import { createClient } from "@/lib/supabase/server";
import type { CustomerStatus, CustomerType, InteractionType } from "./customer";

export type CustomerRow = {
  id: string;
  company_id: string;
  name: string;
  type: CustomerType;
  status: CustomerStatus;
  email: string | null;
  phone: string | null;
  tags: string[];
};

export type InteractionRow = {
  id: string;
  type: InteractionType;
  summary: string;
  occurred_at: string;
};

export async function listCustomers(companyFilter: string | "all"): Promise<CustomerRow[]> {
  const supabase = await createClient();
  let q = supabase
    .from("customers")
    .select("id,company_id,name,type,status,email,phone,tags")
    .order("name");
  if (companyFilter !== "all") q = q.eq("company_id", companyFilter);
  const { data, error } = await q;
  if (error) throw error;
  return (data ?? []) as CustomerRow[];
}

export async function getCustomer(id: string) {
  const supabase = await createClient();
  const { data, error } = await supabase
    .from("customers")
    .select("*")
    .eq("id", id)
    .single();
  if (error) throw error;
  return data;
}

export async function listInteractions(customerId: string): Promise<InteractionRow[]> {
  const supabase = await createClient();
  const { data, error } = await supabase
    .from("interactions")
    .select("id,type,summary,occurred_at")
    .eq("customer_id", customerId)
    .order("occurred_at", { ascending: false });
  if (error) throw error;
  return (data ?? []) as InteractionRow[];
}

export async function crmSummary(companyFilter: string | "all") {
  const customers = await listCustomers(companyFilter);
  const totalCustomers = customers.length;
  const leadCount = customers.filter((c) => c.status === "lead").length;
  const activeCount = customers.filter((c) => c.status === "active").length;
  return { totalCustomers, leadCount, activeCount };
}
```

"use server";
import { createClient } from "@/lib/supabase/server";
import { revalidatePath } from "next/cache";
import { parseTags } from "./customer";

export async function createCustomer(formData: FormData) {
  const supabase = await createClient();
  const { data: { user } } = await supabase.auth.getUser();
  const { data, error } = await supabase.from("customers").insert({
    company_id: String(formData.get("company_id")),
    name: String(formData.get("name")),
    type: formData.get("type") as "particular" | "tienda" | "mayorista",
    status: formData.get("status") as "lead" | "active" | "inactive",
    email: (formData.get("email") as string) || null,
    phone: (formData.get("phone") as string) || null,
    tags: parseTags(String(formData.get("tags") ?? "")),
    notes: (formData.get("notes") as string) || null,
    created_by: user?.id ?? null,
  }).select("id").single();
  if (error) return { error: error.message };
  revalidatePath("/crm");
  return { id: data.id };
}

export async function updateCustomer(id: string, formData: FormData) {
  const supabase = await createClient();
  const { error } = await supabase.from("customers").update({
    name: String(formData.get("name")),
    type: formData.get("type") as "particular" | "tienda" | "mayorista",
    status: formData.get("status") as "lead" | "active" | "inactive",
    email: (formData.get("email") as string) || null,
    phone: (formData.get("phone") as string) || null,
    tags: parseTags(String(formData.get("tags") ?? "")),
    notes: (formData.get("notes") as string) || null,
    updated_at: new Date().toISOString(),
  }).eq("id", id);
  if (error) return { error: error.message };
  revalidatePath(`/crm/${id}`);
  revalidatePath("/crm");
  return { ok: true };
}

export async function logInteraction(customerId: string, formData: FormData) {
  const supabase = await createClient();
  const { data: { user } } = await supabase.auth.getUser();
  const occurredAt = formData.get("occurred_at") as string;
  const { error } = await supabase.from("interactions").insert({
    customer_id: customerId,
    type: formData.get("type") as "llamada" | "email" | "visita" | "whatsapp" | "otro",
    summary: String(formData.get("summary")),
    occurred_at: occurredAt ? new Date(occurredAt).toISOString() : new Date().toISOString(),
    created_by: user?.id ?? null,
  });
  if (error) return { error: error.message };
  revalidatePath(`/crm/${customerId}`);
  return { ok: true };
}
```

"use client";
import { useState } from "react";
import Link from "next/link";
import { Table, TableBody, TableCell, TableHead, TableHeader, TableRow } from "@/components/ui/table";
import { Badge } from "@/components/ui/badge";
import { Input } from "@/components/ui/input";
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@/components/ui/select";
import { statusLabel, typeLabel } from "@/lib/crm/customer";
import type { CustomerRow } from "@/lib/crm/queries";

const STATUS_VARIANT: Record<string, "default" | "secondary" | "outline"> = {
  lead: "secondary",
  active: "default",
  inactive: "outline",
};

export function CustomerTable({ customers }: { customers: CustomerRow[] }) {
  const [search, setSearch] = useState("");
  const [type, setType] = useState("all");
  const [status, setStatus] = useState("all");

  const allTags = Array.from(new Set(customers.flatMap((c) => c.tags))).sort();
  const [tag, setTag] = useState("all");

  const filtered = customers.filter((c) => {
    const q = search.trim().toLowerCase();
    const matchesSearch =
      !q ||
      c.name.toLowerCase().includes(q) ||
      (c.email ?? "").toLowerCase().includes(q) ||
      (c.phone ?? "").toLowerCase().includes(q);
    const matchesType = type === "all" || c.type === type;
    const matchesStatus = status === "all" || c.status === status;
    const matchesTag = tag === "all" || c.tags.includes(tag);
    return matchesSearch && matchesType && matchesStatus && matchesTag;
  });

  return (
    <div className="space-y-4">
      <div className="flex flex-wrap gap-3">
        <Input
          placeholder="Buscar nombre, email o teléfono"
          value={search}
          onChange={(e) => setSearch(e.target.value)}
          className="max-w-xs"
        />
        <Select value={type} onValueChange={setType}>
          <SelectTrigger className="w-40"><SelectValue placeholder="Tipo" /></SelectTrigger>
          <SelectContent>
            <SelectItem value="all">Todos los tipos</SelectItem>
            <SelectItem value="particular">Particular</SelectItem>
            <SelectItem value="tienda">Tienda</SelectItem>
            <SelectItem value="mayorista">Mayorista</SelectItem>
          </SelectContent>
        </Select>
        <Select value={status} onValueChange={setStatus}>
          <SelectTrigger className="w-40"><SelectValue placeholder="Estado" /></SelectTrigger>
          <SelectContent>
            <SelectItem value="all">Todos los estados</SelectItem>
            <SelectItem value="lead">Lead</SelectItem>
            <SelectItem value="active">Activo</SelectItem>
            <SelectItem value="inactive">Inactivo</SelectItem>
          </SelectContent>
        </Select>
        {allTags.length > 0 && (
          <Select value={tag} onValueChange={setTag}>
            <SelectTrigger className="w-40"><SelectValue placeholder="Etiqueta" /></SelectTrigger>
            <SelectContent>
              <SelectItem value="all">Todas las etiquetas</SelectItem>
              {allTags.map((t) => <SelectItem key={t} value={t}>{t}</SelectItem>)}
            </SelectContent>
          </Select>
        )}
      </div>

      {filtered.length === 0 ? (
        <p className="text-sm text-neutral-500">No hay clientes que coincidan.</p>
      ) : (
        <Table>
          <TableHeader>
            <TableRow>
              <TableHead>Nombre</TableHead>
              <TableHead>Tipo</TableHead>
              <TableHead>Estado</TableHead>
              <TableHead>Email</TableHead>
              <TableHead>Teléfono</TableHead>
            </TableRow>
          </TableHeader>
          <TableBody>
            {filtered.map((c) => (
              <TableRow key={c.id}>
                <TableCell>
                  <Link href={`/crm/${c.id}`} className="font-medium hover:underline">{c.name}</Link>
                </TableCell>
