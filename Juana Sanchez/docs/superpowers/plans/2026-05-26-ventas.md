# Módulo Ventas — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Construir el módulo Ventas (registro manual de ventas que descuentan stock, con estados/canal/pago, numeración por empresa/año, cancelación que reingresa stock, y conversión desde cotización aceptada) sobre los cimientos + Inventario + CRM + Cotizaciones.

**Architecture:** Tres tablas Supabase nuevas (`sales`, `sale_items`, `sale_counters`) con RLS por `company_id`, RPC atómico `next_sale_seq` (security definer + guard de empresa) y trigger cliente-misma-empresa. Al crear una venta, la server action inserta `stock_movements` tipo `out` en Inventario (una por línea con producto); al cancelar, inserta `in` (reverso, con guard). Totales y líneas reutilizan la lógica pura y el `line-items-editor` de Cotizaciones (Ventas depende de Cotizaciones). Deploy = push a `main`.

**Tech Stack:** Next.js 15 (App Router, TS), Supabase (`@supabase/ssr`), shadcn/ui + Tailwind, Vitest, Vercel. Todo ya instalado.

**Spec:** `docs/superpowers/specs/2026-05-26-ventas-design.md`

create type public.sale_status    as enum ('pendiente', 'pagada', 'entregada', 'cancelada');
create type public.payment_method as enum ('efectivo', 'tarjeta', 'transferencia', 'bizum', 'otro');
create type public.sale_channel   as enum ('tienda', 'online', 'feria');

create table public.sales (
  id uuid primary key default gen_random_uuid(),
  company_id uuid not null references public.companies(id) on delete cascade,
  customer_id uuid not null references public.customers(id) on delete restrict,
  quote_id uuid references public.quotes(id) on delete set null,
  number text not null,
  seq integer not null,
  year integer not null,
  status public.sale_status not null default 'pendiente',
  sale_date date not null default current_date,
  channel public.sale_channel,
  payment_method public.payment_method,
  tax_rate numeric(5,2) not null default 21,
  notes text,
  subtotal numeric(12,2) not null default 0,
  tax_amount numeric(12,2) not null default 0,
  total numeric(12,2) not null default 0,
  created_by uuid references public.profiles(id),
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),
  unique (company_id, number)
);

create table public.sale_items (
  id uuid primary key default gen_random_uuid(),
  sale_id uuid not null references public.sales(id) on delete cascade,
  product_id uuid references public.products(id) on delete set null,
  description text not null,
  quantity numeric(12,2) not null default 1,
  unit_price numeric(12,2) not null default 0,
  discount_pct numeric(5,2) not null default 0,
  line_subtotal numeric(12,2) not null default 0,
  position integer not null default 0
);

create table public.sale_counters (
  company_id uuid not null references public.companies(id) on delete cascade,
  year integer not null,
  last_seq integer not null default 0,
  primary key (company_id, year)
);

create index on public.sales (company_id);
create index on public.sales (customer_id);
create index on public.sales (quote_id);
create index on public.sale_items (sale_id);

create function public.next_sale_seq(p_company_id uuid, p_year int)
returns int language plpgsql security definer set search_path = public as $$
declare v_seq int;
begin
  if p_company_id not in (select public.accessible_company_ids()) then
    raise exception 'permiso denegado para la empresa %', p_company_id;
  end if;
  insert into public.sale_counters (company_id, year, last_seq)
    values (p_company_id, p_year, 1)
  on conflict (company_id, year)
    do update set last_seq = public.sale_counters.last_seq + 1
  returning last_seq into v_seq;
  return v_seq;
end; $$;

create function public.sales_check_customer_company()
returns trigger language plpgsql security definer set search_path = public as $$
begin
  if not exists (select 1 from public.customers where id = new.customer_id and company_id = new.company_id) then
    raise exception 'El cliente % no pertenece a la empresa %', new.customer_id, new.company_id;
  end if;
  return new;
end; $$;

create trigger sales_customer_company_check
  before insert or update of customer_id, company_id on public.sales
  for each row execute function public.sales_check_customer_company();
```

import { createClient } from "@/lib/supabase/server";
import type { SaleStatus, SaleChannel, PaymentMethod } from "./sale";

export type SaleListRow = {
  id: string;
  number: string;
  status: SaleStatus;
  channel: SaleChannel | null;
  sale_date: string;
  total: number;
  customer: { name: string } | null;
};

export async function listSales(companyFilter: string | "all"): Promise<SaleListRow[]> {
  const supabase = await createClient();
  let q = supabase
    .from("sales")
    .select("id,number,status,channel,sale_date,total,customer:customers(name)")
    .order("sale_date", { ascending: false })
    .order("seq", { ascending: false });
  if (companyFilter !== "all") q = q.eq("company_id", companyFilter);
  const { data, error } = await q;
  if (error) throw error;
  return (data ?? []).map((r) => ({
    id: r.id,
    number: r.number,
    status: r.status as SaleStatus,
    channel: r.channel as SaleChannel | null,
    sale_date: r.sale_date,
    total: r.total,
    customer: r.customer as { name: string } | null,
  }));
}

export async function getSale(id: string) {
  const supabase = await createClient();
  const { data, error } = await supabase
    .from("sales")
    .select("*, customer:customers(name,email,phone), company:companies(name), quote:quotes(number), items:sale_items(*)")
    .eq("id", id)
    .single();
  if (error) throw error;
  return data;
}

export async function salesSummary(companyFilter: string | "all") {
  const sales = await listSales(companyFilter);
  const active = sales.filter((s) => s.status !== "cancelada");
  const totalSales = active.length;
  const totalAmount = active.reduce((sum, s) => sum + s.total, 0);
  const paidAmount = sales.filter((s) => s.status === "pagada").reduce((sum, s) => sum + s.total, 0);
  return { totalSales, totalAmount, paidAmount };
}

export type ConversionLine = {
  product_id: string | null;
  description: string;
  quantity: number;
  unit_price: number;
  discount_pct: number;
};
export type QuoteConversion = {
  customer_id: string;
  tax_rate: number;
  notes: string | null;
  lines: ConversionLine[];
} | null;

export async function getQuoteForConversion(quoteId: string): Promise<QuoteConversion> {
  const supabase = await createClient();
  const { data, error } = await supabase
    .from("quotes")
    .select("customer_id,tax_rate,notes,status,items:quote_items(product_id,description,quantity,unit_price,discount_pct,position)")
    .eq("id", quoteId)
    .single();
  if (error || !data || data.status !== "aceptada") return null;
  const items = ((data.items ?? []) as Array<ConversionLine & { position: number }>)
    .slice()
    .sort((a, b) => a.position - b.position)
    .map((it) => ({
      product_id: it.product_id,
      description: it.description,
      quantity: it.quantity,
      unit_price: it.unit_price,
      discount_pct: it.discount_pct,
    }));
  return { customer_id: data.customer_id as string, tax_rate: data.tax_rate as number, notes: (data.notes as string | null), lines: items };
}

"use server";
import { createClient } from "@/lib/supabase/server";
import { revalidatePath } from "next/cache";
import { lineSubtotal, quoteTotals, companyPrefix } from "@/lib/cotizaciones/quote";
import { formatSaleNumber } from "./sale";

export type SaleLineInput = {
  product_id: string | null;
  description: string;
  quantity: number;
  unit_price: number;
  discount_pct: number;
};

export type SaleInput = {
  company_id: string;
  customer_id: string;
  quote_id: string | null;
  sale_date: string;            // "YYYY-MM-DD"
  channel: "tienda" | "online" | "feria" | null;
  payment_method: "efectivo" | "tarjeta" | "transferencia" | "bizum" | "otro" | null;
  tax_rate: number;
  notes: string | null;
  lines: SaleLineInput[];
};

export async function createSale(input: SaleInput) {
  const supabase = await createClient();
  const { data: { user } } = await supabase.auth.getUser();

  const year = Number(input.sale_date.slice(0, 4));

  const { data: company, error: cErr } = await supabase
    .from("companies").select("name").eq("id", input.company_id).single();
  if (cErr) return { error: cErr.message };

  const { data: seq, error: sErr } = await supabase
    .rpc("next_sale_seq", { p_company_id: input.company_id, p_year: year });
  if (sErr || seq == null) return { error: sErr?.message ?? "No se pudo numerar la venta." };

  const number = formatSaleNumber(companyPrefix(company.name), year, seq);
  const totals = quoteTotals(input.lines, input.tax_rate);

  const { data: sale, error: saleErr } = await supabase.from("sales").insert({
    company_id: input.company_id,
    customer_id: input.customer_id,
    quote_id: input.quote_id,
    number, seq, year,
    sale_date: input.sale_date,
    channel: input.channel,
    payment_method: input.payment_method,
    tax_rate: input.tax_rate,
    notes: input.notes,
    subtotal: totals.subtotal,
    tax_amount: totals.taxAmount,
    total: totals.total,
    created_by: user?.id ?? null,
  }).select("id,number").single();
  if (saleErr) return { error: saleErr.message };

  const itemsRows = input.lines.map((l, i) => ({
    sale_id: sale.id,
    product_id: l.product_id,
    description: l.description,
    quantity: l.quantity,
    unit_price: l.unit_price,
    discount_pct: l.discount_pct,
    line_subtotal: lineSubtotal(l.quantity, l.unit_price, l.discount_pct),
    position: i,
  }));
  if (itemsRows.length > 0) {
    const { error: iErr } = await supabase.from("sale_items").insert(itemsRows);
    if (iErr) {
      await supabase.from("sales").delete().eq("id", sale.id);
      return { error: iErr.message };
    }
  }

  // Bucle con Inventario: descuento de stock (out) por cada línea con producto.
  const movements = input.lines
    .filter((l) => l.product_id)
    .map((l) => ({
      product_id: l.product_id as string,
      type: "out" as const,
      quantity: Math.round(l.quantity),
      reason: "venta",
      note: `Venta ${sale.number}`,
      created_by: user?.id ?? null,
    }));
  if (movements.length > 0) {
    const { error: mErr } = await supabase.from("stock_movements").insert(movements);
    if (mErr) {
      await supabase.from("sales").delete().eq("id", sale.id);
      return { error: mErr.message };
    }
  }

  revalidatePath("/ventas");
  revalidatePath("/inventario");
  return { id: sale.id };
}

export async function setSaleStatus(id: string, status: "pendiente" | "pagada" | "entregada") {
  const supabase = await createClient();
  const { data: sale, error: gErr } = await supabase.from("sales").select("status").eq("id", id).single();
  if (gErr) return { error: gErr.message };
  if (sale.status === "cancelada") return { error: "Una venta cancelada no se puede reactivar." };
  const { error } = await supabase.from("sales")
    .update({ status, updated_at: new Date().toISOString() }).eq("id", id);
  if (error) return { error: error.message };
  revalidatePath(`/ventas/${id}`);
  revalidatePath("/ventas");
  return { ok: true };
}

export async function cancelSale(id: string) {
  const supabase = await createClient();
  const { data: { user } } = await supabase.auth.getUser();
  const { data: sale, error: gErr } = await supabase.from("sales").select("status,number").eq("id", id).single();
  if (gErr) return { error: gErr.message };
  if (sale.status === "cancelada") return { ok: true }; // ya cancelada: no doble reverso

  const { data: items, error: iErr } = await supabase
    .from("sale_items").select("product_id,quantity").eq("sale_id", id);
  if (iErr) return { error: iErr.message };

  const movements = (items ?? [])
    .filter((it) => it.product_id)
    .map((it) => ({
      product_id: it.product_id as string,
      type: "in" as const,
      quantity: Math.round(it.quantity),
      reason: "cancelación de venta",
      note: `Cancelación venta ${sale.number}`,
      created_by: user?.id ?? null,
    }));
  if (movements.length > 0) {
    const { error: mErr } = await supabase.from("stock_movements").insert(movements);
    if (mErr) return { error: mErr.message };
  }

  const { error: uErr } = await supabase.from("sales")
    .update({ status: "cancelada", updated_at: new Date().toISOString() }).eq("id", id);
  if (uErr) return { error: uErr.message };

  revalidatePath(`/ventas/${id}`);
  revalidatePath("/ventas");
  revalidatePath("/inventario");
  return { ok: true };
}
```

"use client";
import { useState } from "react";
import { useRouter } from "next/navigation";
import { Button } from "@/components/ui/button";
import { setSaleStatus, cancelSale } from "@/lib/ventas/actions";
import type { SaleStatus } from "@/lib/ventas/sale";

const NEXT: { status: "pendiente" | "pagada" | "entregada"; label: string }[] = [
  { status: "pendiente", label: "Marcar pendiente" },
  { status: "pagada", label: "Marcar pagada" },
  { status: "entregada", label: "Marcar entregada" },
];

export function SaleStatusActions({ id, current }: { id: string; current: SaleStatus }) {
  const router = useRouter();
  const [pending, setPending] = useState(false);

  if (current === "cancelada") {
    return <p className="text-sm text-neutral-500">Venta cancelada (el stock fue reingresado).</p>;
  }

  async function change(status: "pendiente" | "pagada" | "entregada") {
    if (pending) return;
    setPending(true);
    const res = await setSaleStatus(id, status);
    setPending(false);
    if (res?.error) { alert(res.error); return; }
    router.refresh();
  }
  async function cancel() {
    if (pending) return;
    if (!confirm("¿Cancelar esta venta? El stock de los productos se reingresará.")) return;
    setPending(true);
    const res = await cancelSale(id);
    setPending(false);
    if (res?.error) { alert(res.error); return; }
    router.refresh();
  }

  return (
    <div className="flex flex-wrap gap-2">
      {NEXT.filter((n) => n.status !== current).map((n) => (
        <Button key={n.status} variant="outline" size="sm" disabled={pending} onClick={() => change(n.status)}>{n.label}</Button>
      ))}
      <Button variant="destructive" size="sm" disabled={pending} onClick={cancel}>Cancelar venta</Button>
    </div>
  );
}
